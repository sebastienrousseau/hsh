4//! Bcrypt wrapper with explicit 72-byte safety rail.
5//!
6//! ## Why the safety rail
7//!
8//! Bcrypt silently truncates passwords at 72 bytes. This produced real
9//! authentication-bypass CVEs in 2024–2025:
10//!
11//! - **CVE-2025-22228** — Spring Security `BCryptPasswordEncoder` silently
12//!   accepted passwords `>72` chars as equal to their first-72-byte prefix.
13//! - **CVE-2025-68402** — FreshRSS triggered the same class of bug when an
14//!   unrelated SHA-1 → SHA-256 nonce upgrade pushed the input over 72
15//!   bytes.
16//! - **Okta delegated-auth bypass (Oct 2024)** — cache keys built as
17//!   `bcrypt(SHA1(user+session+pw))` collided when the SHA-1 hex pushed
18//!   bcrypt's input beyond 72 bytes.
19//!
20//! This wrapper **rejects** any password longer than 72 bytes by default.
21//! Callers that genuinely need to support longer inputs must opt in to a
22//! pre-hash via [`BcryptParams::with_prehash`](crate::algorithms::bcrypt::BcryptParams::with_prehash).
23//!
24//! ## Storage format when a pre-hash is configured
25//!
26//! Bcrypt's MCF (`$2b$<cost>$<salt+hash>`) has no parameter slot for a
27//! pre-hash marker, so [`crate::api::hash`] wraps the MCF in the
28//! `hsh-bcrypt-sha256:<mcf>` envelope when
29//! [`PrehashAlgorithm::Sha256`](crate::algorithms::bcrypt::PrehashAlgorithm::Sha256)
30//! is set on the [`crate::policy::Policy`]. The envelope round-trips
31//! through [`crate::api::verify_and_upgrade`], which routes the password
32//! through the same pre-hash before comparing — without the envelope the
33//! verify side would feed bcrypt the raw password and the comparison
34//! would always fail. The envelope also composes with the
35//! `hsh-pepper:<keyver>:` wrapper: peppered + pre-hashed bcrypt hashes
36//! are stored as `hsh-pepper:<keyver>:hsh-bcrypt-sha256:<mcf>`. Pre-hash
37//! mode drift (stored mode ≠ policy mode) triggers an `Outcome::Valid
38//! { rehashed: Some(_) }` on the next successful verify.
39

42use bcrypt::DEFAULT_COST;

45/// Maximum password length bcrypt can handle without silent truncation.
46pub const BCRYPT_MAX_INPUT_BYTES: usize = 72;
47
48/// The highest bcrypt cost factor accepted when *verifying* a stored
49/// hash, unless a caller opts into a different bound.
50///
51/// bcrypt encodes its work factor in the hash string itself, so
52/// verification cost is dictated by the stored value rather than by
53/// local policy. Anything that can influence what gets stored — or that
54/// simply submits a crafted string to a verification endpoint — can
55/// therefore choose how much CPU the check burns. Cost is a base-2
56/// exponent, so the growth is steep. Measured on one x86-64 laptop,
57/// release build, `bcrypt::verify`:
58///
59/// | cost | verify |
60/// |-----:|-------:|
61/// |   10 |  57 ms |
62/// |   12 | 196 ms |
63/// |   14 | 784 ms |
64/// |   16 |  3.0 s |
65/// |   18 | 18.4 s |
66/// |   20 |    73 s |
67///
68/// Fuzzing found `$2x$29$…` — 512x the work of cost 20 — stalling a
69/// single verification for over 29 minutes.
70///
71/// 16 is the default ceiling: it admits every cost a real deployment is
72/// likely to use (OWASP puts the 2025 minimum at 10; high-security
73/// setups reach 12–14) with headroom, while bounding one verification
74/// to a few seconds rather than half an hour. Callers who deliberately
75/// store higher-cost hashes should use
76/// [`Bcrypt::verify_with_max_cost`] rather than raise this.
77pub const MAX_VERIFY_COST: u32 = 16;
78
79/// Reads the cost factor out of a bcrypt hash string.
80///
81/// The format is `$2<variant>$<cost>$<22-char salt><31-char digest>`,
82/// so the cost is the second `$`-delimited field. Returns `None` if the
83/// string is not shaped like a bcrypt hash; callers treat that as "not
84/// ours to reject" and let the underlying implementation report it.
85pub(crate) fn stored_cost(stored: &str) -> Option<u32> {
86    let mut parts = stored.split('$');
87    // A leading `$` means the first field is empty.
88    if parts.next() != Some("") {
89        return None;
90    }
91    let variant = parts.next()?;
92    if !matches!(variant, "2" | "2a" | "2b" | "2x" | "2y") {
93        return None;
94    }
95    parts.next()?.parse().ok()
96}

98/// Pre-hash algorithm to apply when the password exceeds 72 bytes.
99#[derive(
100    Clone,
101    Copy,
102    Debug,
103    Default,
104    Eq,
105    Hash,
106    Ord,
107    PartialEq,
108    PartialOrd,
109    Serialize,
110    Deserialize,
111)]
112pub enum PrehashAlgorithm {
113    /// No pre-hash — bcrypt receives the password verbatim and rejects
114    /// inputs `>72` bytes. **Recommended default.**
115    #[default]
116    None,
117    /// Hash the password with HMAC-SHA-256 keyed by the bcrypt salt
118    /// before passing the 32-byte digest to bcrypt. Lets you accept
119    /// arbitrary-length inputs without truncation.
120    Sha256,
121}
122
123/// Bcrypt parameters.
124#[derive(Clone, Copy, Debug, Serialize, Deserialize)]
125pub struct BcryptParams {
126    /// Bcrypt cost factor (log2 of work). OWASP-2025 minimum is 10.
127    pub cost: u32,
128    /// Optional pre-hash to allow inputs longer than 72 bytes.
129    pub prehash: PrehashAlgorithm,
130}
131
132impl Default for BcryptParams {
133    fn default() -> Self {
134        Self {
135            cost: DEFAULT_COST,
136            prehash: PrehashAlgorithm::None,
137        }
138    }
139}
140
141impl BcryptParams {
142    /// Builds a bcrypt parameter set with the given cost factor.
143    pub fn new(cost: u32) -> Self {
144        Self {
145            cost,
146            prehash: PrehashAlgorithm::None,
147        }
148    }
149
150    /// Enables the pre-hash safety adapter so passwords longer than
151    /// 72 bytes are accepted via HMAC-SHA-256 pre-hash.
152    pub fn with_prehash(mut self, algo: PrehashAlgorithm) -> Self {
153        self.prehash = algo;
154        self
155    }
156}

172
173impl HashingAlgorithm for Bcrypt {
174    /// Hashes `password` using bcrypt at [`DEFAULT_COST`].
175    ///
176    /// The `_salt` argument is ignored — bcrypt generates its own salt.
177    /// Inputs longer than [`BCRYPT_MAX_INPUT_BYTES`] are **rejected** with
178    /// [`Error::InvalidPassword`]; use [`Bcrypt::hash_with`] for an opt-in
179    /// pre-hash policy.
180    fn hash_password(password: &str, _salt: &str) -> Result<Vec<u8>> {
181        Self::hash_with(password, BcryptParams::default())
182    }
183}
184
185impl Bcrypt {
186    /// Hashes `password` under explicit [`BcryptParams`].
187    pub fn hash_with(
188        password: &str,
189        params: BcryptParams,
190    ) -> Result<Vec<u8>> {
191        let payload =
192            prepare_payload(password.as_bytes(), params.prehash)?;
193        bcrypt::hash(&payload, params.cost)
194            .map(String::into_bytes)
195            .map_err(|e| {
196                Error::hashing(
197                    crate::error::HashingErrorKind::Bcrypt,
198                    e.to_string(),
199                )
200            })
201    }
202
203    /// Verifies `password` against a bcrypt hash string.
204    ///
205    /// Constant-time comparison is delegated to the `bcrypt` crate,
206    /// which uses `subtle` internally.
207    pub fn verify(
208        password: &str,
209        stored: &str,
210        prehash: PrehashAlgorithm,
211    ) -> Result<bool> {
212        Self::verify_with_max_cost(
213            password,
214            stored,
215            prehash,
216            MAX_VERIFY_COST,
217        )
218    }
219
220    /// Verifies `password` against a bcrypt hash string, refusing any
221    /// stored cost above `max_cost`.
222    ///
223    /// [`verify`](Self::verify) delegates here with
224    /// [`MAX_VERIFY_COST`]. Use this directly only when the stored
225    /// hashes are known to carry a higher work factor by design, and
226    /// keep the bound as low as those hashes allow: it is the only
227    /// thing standing between a verification endpoint and an
228    /// attacker-chosen amount of CPU.
229    ///
230    /// # Errors
231    ///
232    /// Returns [`Error::InvalidHashString`] if the stored hash
233    /// advertises a cost above `max_cost`, before any work is done, and
234    /// [`Error::Verification`] if the underlying check fails.
235    pub fn verify_with_max_cost(
236        password: &str,
237        stored: &str,
238        prehash: PrehashAlgorithm,
239        max_cost: u32,
240    ) -> Result<bool> {
241        // Check the advertised work factor before doing any of it. The
242        // cost lives in the stored string, so without this an untrusted
243        // value decides how long verification runs.
244        if let Some(cost) = stored_cost(stored) {
245            if cost > max_cost {
246                return Err(Error::InvalidHashString(
247                    format!(
248                        "bcrypt cost {cost} exceeds the maximum accepted for verification ({max_cost})"
249                    )
250                    .into(),
251                ));
252            }
253        }
254
255        let payload = prepare_payload(password.as_bytes(), prehash)?;
256        bcrypt::verify(&payload, stored).map_err(|_| {
257            Error::Verification("bcrypt verify failed".into())
258        })
259    }
260}
261
262fn prepare_payload(
263    password: &[u8],
264    prehash: PrehashAlgorithm,
265) -> Result<Vec<u8>> {
266    match prehash {
267        PrehashAlgorithm::None => {
268            if password.len() > BCRYPT_MAX_INPUT_BYTES {
269                return Err(Error::InvalidPassword(
270                    "bcrypt input exceeds 72 bytes; opt into a pre-hash via BcryptParams::with_prehash to handle longer inputs".into(),
271                ));
272            }
273            Ok(password.to_vec())
274        }
275        PrehashAlgorithm::Sha256 => {
276            use sha2::{Digest, Sha256};
277            let digest = Sha256::digest(password);
278            // bcrypt's input must be valid UTF-8; b64-encode the digest
279            // to ensure that without losing entropy.
280            use base64::{engine::general_purpose, Engine as _};
281            Ok(general_purpose::STANDARD_NO_PAD
282                .encode(digest)
283                .into_bytes())
284        }
285    }
286}

288#[cfg(test)]
289mod verify_cost_tests {
290    //! Unit tests for the cost guard's parsing half.
291    //!
292    //! Anything that passes a password to `Bcrypt::verify` lives in
293    //! `tests/test_bcrypt_cost_bound.rs` instead. `crates/*/src/` is
294    //! analysed by CodeQL as production code, where a hard-coded
295    //! password is a critical finding; `crates/*/tests/**` is excluded
296    //! precisely because fixtures there are expected to carry them.
297    //! See `.github/codeql/codeql-config.yml`.
298
299    use super::*;
300
301    /// The cost carried by the input libFuzzer found stalling
302    /// `fuzz_phc_parse` for over 29 minutes.
303    const FUZZ_TIMEOUT_COST: u32 = 29;
304
305    #[test]
306    fn reads_the_cost_from_each_variant_prefix() {
307        for prefix in ["2", "2a", "2b", "2x", "2y"] {
308            let stored = format!("${prefix}$12$abcdefghijklmnopqrstuv");
309            assert_eq!(
310                stored_cost(&stored),
311                Some(12),
312                "prefix {prefix}"
313            );
314        }
315    }
316
317    #[test]
318    fn ignores_strings_that_are_not_bcrypt() {
319        assert_eq!(
320            stored_cost("$argon2id$v=19$m=8,t=1,p=1$c2FsdA$aGFzaA"),
321            None
322        );
323        assert_eq!(stored_cost("not-a-hash"), None);
324        assert_eq!(stored_cost(""), None);
325        assert_eq!(stored_cost("$2b$notanumber$abc"), None);
326    }
327
328    #[test]
329    fn the_fuzz_input_cost_is_above_the_default_bound() {
330        let stored = format!(
331            "$2x${FUZZ_TIMEOUT_COST}$rrjrrdrjrrdrh..jrrdrh..n....jrrdrh..."
332        );
333        let cost = stored_cost(&stored).expect("bcrypt string");
334        assert_eq!(cost, FUZZ_TIMEOUT_COST);
335        assert!(cost > MAX_VERIFY_COST, "must be refused by default");
336        assert!(
337            cost <= 31,
338            "must be admitted once the bound is raised"
339        );
340    }
341
342    #[test]
343    fn the_bound_is_inclusive_at_the_ceiling() {
344        let at =
345            format!("$2b${MAX_VERIFY_COST}$abcdefghijklmnopqrstuv");
346        let over = format!(
347            "$2b${}$abcdefghijklmnopqrstuv",
348            MAX_VERIFY_COST + 1
349        );
350        assert_eq!(stored_cost(&at), Some(MAX_VERIFY_COST));
351        assert_eq!(stored_cost(&over), Some(MAX_VERIFY_COST + 1));
352    }
